int lwline=94;
int lw=119;
int xxx=0;

void get_highest(void)
{
    lwline=94;
    lw=119;
    xxx=0;

    for (int i=64;i<124;i+=3)          //从64列到124列从下往上找最长白列，隔三列扫一次减少代码运行时间
    {
        if(image_use[80][i]>=daj)    //每一列的80行为黑就不扫
        {
            for (int j=85;j>=0;j--)
            {
                //找到白黑黑跳变，消除噪点影响
                if ((image_use[j][i]>=daj&&image_use[j-1][i]<daj&&image_use[j-2][i]<daj)||j<=5)
                {
                    //贪心算法求得最长白列
                    if (j<lw)
                    {
                      lw=j;
                      lwline=i;
                    }
                    break;
                }
            }
        }
    }
}
